# ============================================================
# AIR QUALITY PREDICTION USING MACHINE LEARNING
# ============================================================

# Import required libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error
from sklearn.metrics import mean_squared_error
from sklearn.metrics import r2_score


# ============================================================
# 1. LOAD DATASET
# ============================================================

df = pd.read_csv(
    "AirQualityUCI.csv",
    sep=";",
    decimal=","
)

print("Dataset loaded successfully!")

print("Dataset Shape:")
print(df.shape)

print("\nFirst 5 Rows:")
print(df.head())


# ============================================================
# 2. REMOVE EMPTY COLUMNS
# ============================================================

df = df.dropna(axis=1, how="all")

print("\nShape after removing empty columns:")
print(df.shape)


# ============================================================
# 3. DISPLAY COLUMN NAMES
# ============================================================

print("\nColumn Names:")
print(df.columns.tolist())


# ============================================================
# 4. REPLACE MISSING VALUES
# ============================================================

# In the UCI Air Quality dataset,
# -200 represents missing values.

df.replace(-200, np.nan, inplace=True)

print("\nMissing Values:")
print(df.isnull().sum())


# ============================================================
# 5. CREATE DATETIME COLUMN
# ============================================================

df["Datetime"] = pd.to_datetime(
    df["Date"] + " " + df["Time"],
    dayfirst=True,
    errors="coerce"
)

print("\nDatetime column created successfully.")


# ============================================================
# 6. SORT DATA BY DATETIME
# ============================================================

df.sort_values(
    "Datetime",
    inplace=True
)

df.reset_index(
    drop=True,
    inplace=True
)


# ============================================================
# 7. CREATE TIME FEATURES
# ============================================================

df["Hour"] = df["Datetime"].dt.hour

df["Day"] = df["Datetime"].dt.day

df["Month"] = df["Datetime"].dt.month

df["DayOfWeek"] = df["Datetime"].dt.dayofweek


print("\nTime features created:")
print("Hour")
print("Day")
print("Month")
print("DayOfWeek")


# ============================================================
# 8. HANDLE MISSING VALUES
# ============================================================

numeric_columns = df.select_dtypes(
    include=np.number
).columns

# Interpolation
df[numeric_columns] = df[numeric_columns].interpolate(
    method="linear"
)

# Fill remaining missing values using median
df[numeric_columns] = df[numeric_columns].fillna(
    df[numeric_columns].median()
)

print("\nMissing values after preprocessing:")
print(df.isnull().sum())


# ============================================================
# 9. STATISTICAL SUMMARY
# ============================================================

print("\nStatistical Summary:")
print(df.describe())


# ============================================================
# 10. EXPLORATORY DATA ANALYSIS
# ============================================================

# ------------------------------------------------------------
# CO DISTRIBUTION
# ------------------------------------------------------------

plt.figure(figsize=(8, 5))

plt.hist(
    df["CO(GT)"],
    bins=30,
    color="skyblue",
    edgecolor="black"
)

plt.xlabel("CO Concentration")
plt.ylabel("Frequency")

plt.title(
    "Distribution of CO Concentration"
)

plt.tight_layout()

plt.savefig(
    "co_distribution.png"
)

plt.show()


# ============================================================
# 11. CORRELATION HEATMAP
# ============================================================

plt.figure(figsize=(14, 10))

correlation = df.corr(
    numeric_only=True
)

sns.heatmap(
    correlation,
    cmap="coolwarm",
    annot=False
)

plt.title(
    "Correlation Heatmap of Air Quality Features"
)

plt.tight_layout()

plt.savefig(
    "correlation_heatmap.png"
)

plt.show()


# ============================================================
# 12. CO CONCENTRATION OVER TIME
# ============================================================

plt.figure(figsize=(14, 5))

plt.plot(
    df["Datetime"],
    df["CO(GT)"],
    color="blue"
)

plt.xlabel("Date and Time")

plt.ylabel("CO Concentration")

plt.title(
    "CO Concentration Over Time"
)

plt.tight_layout()

plt.savefig(
    "co_over_time.png"
)

plt.show()


# ============================================================
# 13. SELECT TARGET VARIABLE
# ============================================================

# Target variable
target = "CO(GT)"


# ============================================================
# 14. SELECT INPUT FEATURES
# ============================================================

X = df.drop(
    columns=[
        target,
        "Datetime"
    ],
    errors="ignore"
)

y = df[target]


# ============================================================
# 15. REMOVE ROWS WITH MISSING TARGET
# ============================================================

valid_rows = y.notna()

X = X.loc[
    valid_rows
]

y = y.loc[
    valid_rows
]


# ============================================================
# 16. TRAIN-TEST SPLIT
# ============================================================

# 80% Training data
# 20% Testing data

split_index = int(
    len(X) * 0.80
)

X_train = X.iloc[
    :split_index
]

X_test = X.iloc[
    split_index:
]

y_train = y.iloc[
    :split_index
]

y_test = y.iloc[
    split_index:
]


print("\nTraining samples:")
print(len(X_train))

print("\nTesting samples:")
print(len(X_test))


# ============================================================
# 17. CREATE RANDOM FOREST MODEL
# ============================================================

model = RandomForestRegressor(
    n_estimators=100,
    random_state=42,
    n_jobs=-1
)


# ============================================================
# 18. TRAIN MODEL
# ============================================================

print("\nTraining Random Forest model...")

model.fit(
    X_train,
    y_train
)

print("Model training completed!")


# ============================================================
# 19. MAKE PREDICTIONS
# ============================================================

y_pred = model.predict(
    X_test
)


# ============================================================
# 20. PERFORMANCE EVALUATION
# ============================================================

mae = mean_absolute_error(
    y_test,
    y_pred
)

mse = mean_squared_error(
    y_test,
    y_pred
)

rmse = np.sqrt(
    mse
)

r2 = r2_score(
    y_test,
    y_pred
)


# ============================================================
# 21. DISPLAY MODEL PERFORMANCE
# ============================================================

print("\n==========================================")
print("       MODEL PERFORMANCE")
print("==========================================")

print(
    "Mean Absolute Error (MAE):",
    mae
)

print(
    "Mean Squared Error (MSE):",
    mse
)

print(
    "Root Mean Squared Error (RMSE):",
    rmse
)

print(
    "R2 Score:",
    r2
)

print("==========================================")


# ============================================================
# 22. ACTUAL VS PREDICTED GRAPH
# ============================================================

plt.figure(figsize=(14, 6))

plt.plot(
    y_test.values,
    label="Actual",
    color="blue"
)

plt.plot(
    y_pred,
    label="Predicted",
    color="red"
)

plt.xlabel(
    "Test Sample"
)

plt.ylabel(
    "CO Concentration"
)

plt.title(
    "Actual vs Predicted CO Concentration"
)

plt.legend()

plt.tight_layout()

plt.savefig(
    "actual_vs_predicted.png"
)

plt.show()


# ============================================================
# 23. FEATURE IMPORTANCE
# ============================================================

feature_importance = pd.Series(
    model.feature_importances_,
    index=X.columns
)

feature_importance = feature_importance.sort_values(
    ascending=False
)

print("\nFeature Importance:")
print(feature_importance)


# ============================================================
# 24. FEATURE IMPORTANCE GRAPH
# ============================================================

plt.figure(figsize=(10, 6))

feature_importance.head(10).plot(
    kind="bar",
    color="green"
)

plt.xlabel(
    "Features"
)

plt.ylabel(
    "Importance"
)

plt.title(
    "Top 10 Important Features"
)

plt.tight_layout()

plt.savefig(
    "feature_importance.png"
)

plt.show()


# ============================================================
# 25. SAVE PREDICTION RESULTS
# ============================================================

results = pd.DataFrame({
    "Actual": y_test.values,
    "Predicted": y_pred
})

results.to_csv(
    "prediction_results.csv",
    index=False
)

print(
    "\nPrediction results saved as prediction_results.csv"
)


# ============================================================
# 26. FINAL MESSAGE
# ============================================================

print("\n==========================================")
print(" AIR QUALITY PREDICTION COMPLETED")
print("==========================================")
